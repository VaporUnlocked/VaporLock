cmake_minimum_required(VERSION 3.25)
project(AimTestingSuite VERSION 1.1.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(USE_MSVC_RUNTIME_LIBRARY_DLL OFF CACHE BOOL "" FORCE)
set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
set(USE_AUDIO OFF CACHE BOOL "" FORCE)
set(SUPPORT_MODULE_RAUDIO OFF CACHE BOOL "" FORCE)
set(SUPPORT_GPU_SKINNING OFF CACHE BOOL "" FORCE)
set(SUPPORT_CUSTOM_FRAME_CONTROL OFF CACHE BOOL "" FORCE)
set(SUPPORT_SCREEN_CAPTURE OFF CACHE BOOL "" FORCE)
set(SUPPORT_AUTOMATION_EVENTS OFF CACHE BOOL "" FORCE)
set(SUPPORT_CLIPBOARD_IMAGE OFF CACHE BOOL "" FORCE)
set(SUPPORT_BUSY_WAIT_LOOP OFF CACHE BOOL "" FORCE)
set(SUPPORT_PARTIALBUSY_WAIT_LOOP OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/raylib-6.0 EXCLUDE_FROM_ALL)
configure_file(source/bundled_assets.rc.in "${CMAKE_BINARY_DIR}/bundled_assets.rc" @ONLY)
set(SUITE_BUNDLED_SOURCES source/bundled_assets.cpp "${CMAKE_BINARY_DIR}/bundled_assets.rc")
add_executable(AimTestingSuite source/main.cpp source/storage.cpp source/platform.cpp ${SUITE_BUNDLED_SOURCES})
target_link_libraries(AimTestingSuite PRIVATE raylib shell32 ole32)
target_include_directories(AimTestingSuite PRIVATE vendor/raylib-6.0/src vendor/raylib-6.0/src/external/glfw/include)
target_compile_definitions(AimTestingSuite PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
set(SUITE_EXECUTABLE_NAME "Aim Testing Suite" CACHE STRING "Executable filename without extension")
set_target_properties(AimTestingSuite PROPERTIES OUTPUT_NAME "${SUITE_EXECUTABLE_NAME}" RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_SOURCE_DIR}>")
if(MSVC)
  target_compile_options(AimTestingSuite PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/EHsc;/utf-8>)
  target_link_options(AimTestingSuite PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
endif()
enable_testing()
add_executable(SuiteTests source/tests.cpp source/storage.cpp source/platform.cpp ${SUITE_BUNDLED_SOURCES})
target_link_libraries(SuiteTests PRIVATE raylib shell32 ole32)
target_include_directories(SuiteTests PRIVATE vendor/raylib-6.0/src)
target_compile_definitions(SuiteTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
set_target_properties(SuiteTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/build")
if(MSVC)
  target_compile_options(SuiteTests PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
endif()
add_test(NAME SuiteCore COMMAND SuiteTests "${CMAKE_SOURCE_DIR}")
